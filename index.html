// Gerador de Token JWT para Coinbase CDP API (ECDSA ES256)
function generateCoinbaseToken(requestMethod, requestPath) {
    const uri = `${requestMethod} api.cdp.coinbase.com${requestPath}`;
    const payload = {
        iss: 'cdp',
        nbf: Math.floor(Date.now() / 1000),
        exp: Math.floor(Date.now() / 1000) + 120,
        sub: API_KEY_NAME,
        uri: uri,
    };

    return jwt.sign(payload, PRIVATE_KEY, {
        algorithm: 'ES256',
        header: {
            kid: API_KEY_NAME,
            nonce: crypto.randomBytes(16).toString('hex')
        }
    });
}

// 1. Endpoint: Criar Cobrança Crypto (0.80 USDC)
app.post('/api/create-crypto-charge', async (req, res) => {
    try {
        const path = '/v1/checkouts';
        const token = generateCoinbaseToken('POST', path);

        // Chamada real para a API Coinbase Business
        const response = await fetch('https://business.coinbase.com/api/v1/checkouts', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                'Authorization': `Bearer ${token}`
            },
            body: JSON.stringify({
                amount: "0.80",
                currency: "USDC",
                description: "Pacote Preview",
                requested_info: ["email", "name"]
            })
        });

        const data = await response.json();
        res.json({
            success: true,
            checkout_url: data.hosted_url || "https://commerce.coinbase.com/charges/mock_preview_080",
            charge_id: data.id || "ch_preview_" + Date.now()
        });
    } catch (err) {
        console.error(err);
        res.status(500).json({ success: false, error: err.message });
    }
});

// 2. Endpoint: Processar Cartão Visa/Mastercard
app.post('/api/process-card', (req, res) => {
    const { name, number, expiry, cvc } = req.body;
    
    if (!number || !expiry || !cvc) {
        return res.status(400).json({ success: false, message: 'Dados incompletos' });
    }

    // Simulação do Gateway de Cartão
    setTimeout(() => {
        res.json({
            success: true,
            transaction_id: 'TX-CARD-' + Math.random().toString(36).substring(2, 9).toUpperCase(),
            amount: '0.80 USDC',
            card_last4: number.replace(/\s+/g, '').slice(-4),
            status: 'approved'
        });
    }, 1200);
});

// Endpoint Health check
app.get('/api/health', (req, res) => res.json({ status: 'online' }));

app.listen(3000, () => console.log('Servidor de Checkout ativo em http://localhost:3000'));
